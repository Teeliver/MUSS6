find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(MYSQLCLIENT QUIET IMPORTED_TARGET mysqlclient)
endif()
add_library(mu_mysql STATIC query_manager.cpp sql_dialect.cpp)
target_compile_features(mu_mysql PUBLIC cxx_std_17)
target_include_directories(mu_mysql PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mu_mysql PUBLIC mu_platform)
if(MYSQLCLIENT_FOUND)
  target_compile_definitions(mu_mysql PRIVATE MU_HAVE_MYSQL=1)
  target_link_libraries(mu_mysql PUBLIC PkgConfig::MYSQLCLIENT)
else()
  target_compile_definitions(mu_mysql PRIVATE MU_HAVE_MYSQL=0)
  message(WARNING "libmysqlclient unavailable: native QueryManager will fail explicitly; no fake DB backend")
endif()
if(BUILD_TESTING)
  add_executable(mu_mysql_unit_tests tests.cpp)
  target_link_libraries(mu_mysql_unit_tests PRIVATE mu_mysql)
  add_test(NAME mu_mysql_unit_tests COMMAND mu_mysql_unit_tests)
endif()

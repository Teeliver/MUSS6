function(mu_support_target target folder join)
 set(root "${CMAKE_CURRENT_LIST_DIR}/../../${folder}")
 file(GLOB sources CONFIGURE_DEPENDS "${root}/*.cpp")
 foreach(excluded ${folder}.cpp SocketManager.cpp SocketManagerUdp.cpp QueryManager.cpp ServerDisplayer.cpp MiniDump.cpp stdafx.cpp Queue.cpp)
  list(REMOVE_ITEM sources "${root}/${excluded}")
 endforeach()
 add_executable(${target} ${sources} main.cpp socket_manager.cpp display.cpp)
 target_include_directories(${target} PRIVATE "${root}")
 target_link_libraries(${target} PRIVATE mu_platform mu_mysql)
 if(join)
  target_sources(${target} PRIVATE join_udp.cpp)
  target_compile_definitions(${target} PRIVATE MU_NATIVE_JOIN=1 JOINSERVER_UPDATE=603 JOINSERVER_LANGUAGE=1)
 else()
  target_compile_definitions(${target} PRIVATE DATASERVER_UPDATE=603 DATASERVER_LANGUAGE=1 DATASERVER_SHOP=3)
 endif()
endfunction()

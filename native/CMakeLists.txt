cmake_minimum_required(VERSION 3.20)
project(MuOnlineEX603Native LANGUAGES C CXX)
if(NOT APPLE)
 message(FATAL_ERROR "The native port currently targets macOS.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
add_compile_definitions(MU_NATIVE_MACOS=1)
add_compile_options(-fms-extensions -Wno-writable-strings -Wno-unknown-pragmas)
find_package(Threads REQUIRED)
add_library(mu_platform STATIC src/platform.cpp)
target_include_directories(mu_platform PUBLIC include)
target_link_libraries(mu_platform PUBLIC Threads::Threads)
include(CTest)
if(BUILD_TESTING)
 add_executable(mu_platform_contract platform_contract_test.cpp)
 target_link_libraries(mu_platform_contract PRIVATE mu_platform)
 target_compile_options(mu_platform_contract PRIVATE -UNDEBUG)
 add_test(NAME mu_platform_contract COMMAND mu_platform_contract)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lua-5.2.4/src/lua.h")
 add_subdirectory(vendor)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/connect/CMakeLists.txt")
 add_subdirectory(connect)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/db/CMakeLists.txt")
 add_subdirectory(db)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/support/CMakeLists.txt")
 add_subdirectory(support)
endif()
option(MU_BUILD_GAME_LOGIC "Compile original EX603 gameplay modules while runtime is being ported" ON)
if(MU_BUILD_GAME_LOGIC AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/game/CMakeLists.txt")
 add_subdirectory(game)
endif()
add_subdirectory(client)
